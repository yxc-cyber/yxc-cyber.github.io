\section{Research Experience}
\researchentry{Department of Data Science, College of William \& Mary}{Aug 2026 -- Present}
              {Graduate Research Assistant $\vert$ Advisor: Prof. Qingyun Wang}{}
\begin{itemize}
  \item Investigating \textbf{agentic AI and AI for science}, with a focus on reliable AI systems for scientific knowledge acquisition, reasoning, and discovery.

\researchentry{Siebel School of Computing and Data Science, UIUC}{Aug 2024 -- May 2026}
              {Graduate Research Assistant $\vert$ Advisors: Prof. Gokhan Tur and Prof. Dilek Hakkani-Tür}{}
\begin{itemize}
  \item Developed \textbf{Gradient-Based Connections (GBC)}, a credit-assignment method for optimizing prompts in LLM multi-agent systems; released the method in \href{https://github.com/yxc-cyber/AgentChord}{\textbf{AgentChord}}, an open-source multi-agent framework.
  \item Designed a prefix-based technique to reduce credit-assignment space complexity to make GBC practically usable; achieved \textbf{99.0\% Inform, 94.0\% Success, and 91.4\% slot F1} on MultiWOZ and \textbf{24.3\%} on $\tau$-bench, outperforming strong single-agent baselines.
  \item Adapted ReAct to task-oriented dialogue and designed friction-aware prompts to reduce interaction ambiguity, improving MultiWOZ Inform and Success by \textbf{5.5 and 3.0 points}.
  \item Built LLM-based question-generation and student-simulation systems for K--2 literacy assessment; achieved \textbf{50.4\% MAP@1} with ROUGE-L F1 on FairytaleQA, outperforming prior methods.
\end{itemize}

\researchentry{Department of Computer Science, NYU Shanghai}{Jun 2023 -- May 2024}
              {Undergraduate Research Assistant $\vert$ Advisor: Prof. Yik-Cheung (Wilson) Tam}{}
\begin{itemize}
  \item Fine-tuned LoRA adapters for LLM arithmetic reasoning; compared chain-of-thought, Prolog generation, and hybrid outputs, and open-sourced the \textbf{GSM8K-Prolog} dataset.
  \item Evaluated Prolog-permutation data augmentation, improving accuracy over the chain-of-thought baseline by \textbf{10.9 points on GSM8K} and \textbf{22.6 points on GSM-HARD}; analyzed divergence between cross-entropy loss and executable-code accuracy.
  \item Fine-tuned LLMs for dialogue state tracking across model, data, LoRA, and output-format settings, achieving \textbf{82.4\% joint goal accuracy}, a new state of the art among end-to-end DST methods.
\end{itemize}
